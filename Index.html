☕ Java 2D Array Operations

Matrix Addition • Subtraction • Multiplication • Division

📊 Input Matrices

Matrix A


8 6
4 2

Matrix B


4 2
2 1

💻 Java Code

public class TwoDArrayOperations {

    public static void main(String[] args) {

        int[][] matrixA = {
            {8, 6},
            {4, 2}
        };

        int[][] matrixB = {
            {4, 2},
            {2, 1}
        };

        int rows = matrixA.length;
        int cols = matrixA[0].length;

        int[][] sum = new int[rows][cols];
        int[][] sub = new int[rows][cols];
        double[][] div = new double[rows][cols];
        int[][] mul = new int[rows][cols];

        // Addition, Subtraction and Division
        for (int i = 0; i < rows; i++) {

            for (int j = 0; j < cols; j++) {

                sum[i][j] =
                    matrixA[i][j] + matrixB[i][j];

                sub[i][j] =
                    matrixA[i][j] - matrixB[i][j];

                div[i][j] =
                    (double) matrixA[i][j] / matrixB[i][j];
            }
        }

        // Matrix Multiplication
        for (int i = 0; i < rows; i++) {

            for (int j = 0; j < cols; j++) {

                for (int k = 0; k < cols; k++) {

                    mul[i][j] +=
                        matrixA[i][k] * matrixB[k][j];
                }
            }
        }

        // Display results
        System.out.println("--- Addition ---");

        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                System.out.print(sum[i][j] + " ");
            }
            System.out.println();
        }

        System.out.println("--- Subtraction ---");

        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                System.out.print(sub[i][j] + " ");
            }
            System.out.println();
        }

        System.out.println("--- Multiplication ---");

        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                System.out.print(mul[i][j] + " ");
            }
            System.out.println();
        }

        System.out.println("--- Division ---");

        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                System.out.printf("%.2f ", div[i][j]);
            }
            System.out.println();
        }
    }
}
